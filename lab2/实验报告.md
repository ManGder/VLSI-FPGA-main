int readBenchMarkFile(std::string i_file_name){

    std::fstream f;
    f.open(i_file_name, std::ios::in);

    if (!f.is_open()){
        std::printf("file %s open failed", i_file_name.c_str());
        return -1;
    }

    std::string line;
    while (std::getline(f, line)){
        if (line.empty())
            break;
        
        std::istringstream iss(line);
        std::string temp;
        std::vector<std::string> row;
        while (iss >> temp){
            row.push_back(temp);
        }
        if (row.size() == 2){
            int l_size_x = std::stoi(row[0]);
            int l_size_y = std::stoi(row[1]);
            glb_fpga.setSize(l_size_x, l_size_y);
            glb_fpga.initialize();
        }else if (row.size() == 3){
            int l_inst_id, l_x, l_y;
            l_inst_id = std::stoi(row[0]);
            l_x = std::stoi(row[1]);
            l_y = std::stoi(row[2]);
            Instance* inst = new Instance(l_x, l_y, l_inst_id, true);
            glb_inst_map[l_inst_id] = inst;
            glb_fpga.addInst(l_x, l_y, inst);
        }else{
            std::printf("something wrong when try to parser: %s", line.c_str());
            return -1;
        }
    }
    while (std::getline(f, line)){
        if (line.empty())
            continue;
        
        std::istringstream iss(line);
        std::string temp;
        std::vector<std::string> row;
        while (iss >> temp){
            row.push_back(temp);
        }
        int l_inst_id, l_net_id;
        l_inst_id = std::stoi(row[0]);
        Instance* l_inst_point = nullptr;
        if (glb_inst_map.find(l_inst_id) == glb_inst_map.end()){
            l_inst_point = new Instance();
            l_inst_point->setInstId(l_inst_id);
            glb_inst_map[l_inst_id] = l_inst_point;
        }else{
            l_inst_point = glb_inst_map[l_inst_id];
        }
        
        for (size_t i = 1; i < row.size(); i++){
            l_net_id = std::stoi(row[i]);
            Net* lo_net_point = nullptr;
            if (glb_net_map.find(l_net_id) == glb_net_map.end()){
                lo_net_point = new Net;
                lo_net_point->setNetId(l_net_id);
                glb_net_map[l_net_id] = lo_net_point;
            }else{
                lo_net_point = glb_net_map[l_net_id];
            }
            l_inst_point->addNet(lo_net_point);
            lo_net_point->addInst(l_inst_point);
        }
    }
    f.close();
    return 0;
}

int reportWireLength(){

    for (auto lo_net : glb_net_map){
        l_wirelength += lo_net.second->evalHPWL();
    }
    std::cout << "Wirelength: " << std::setw(5) << std::right << l_wirelength << std::endl;

int reportValid(){
    // 检查布局是否合法
    // 1. 检查每个inst的布局位置是否和Block包含的inst一致
    // 2. 检查每个Block是否存在inst重复出现的情况

    int l_error_count = 0;
    // 先检查每个inst的布局位置是否和Block包含的inst一致
    for (auto lo_inst : glb_inst_map){
        Instance* lo_inst_p = lo_inst.second;
        std::pair<int, int> lo_inst_pos = lo_inst_p->getPosition();
        Block* lo_block_p = glb_fpga.getBlock(lo_inst_pos.first, lo_inst_pos.second);
        if (lo_block_p == nullptr){
            std::printf("[ERROR] inst %d is not placed (%d, %d)\n", lo_inst_p->getInstId(), lo_inst_pos.first, lo_inst_pos.second);
            l_error_count++;
            continue;
        }
        if (lo_block_p->getInsts().empty() || lo_block_p->getInsts()[0] != lo_inst_p){
            std::printf("[ERROR] inst %d is not in block (%d, %d)\n", lo_inst_p->getInstId(), lo_inst_pos.first, lo_inst_pos.second);
            l_error_count++;
        }
    }
    // 再从block一侧检查是否存在inst重复出现的情况
    std::set<Instance*> lo_inst_attend;
    for (int i = 0; i < glb_fpga.getSizeX(); i++){
        for (int j = 0; j < glb_fpga.getSizeY(); j++){
            Block* lo_block_p = glb_fpga.getBlock(i, j);
            if (lo_block_p == nullptr)
                continue;
            for (auto lo_inst : lo_block_p->getInsts()){
                if (lo_inst == nullptr)
                    continue;
                if (lo_inst_attend.find(lo_inst) != lo_inst_attend.end()){
                    std::printf("[ERROR] inst %d is repeated in block (%d, %d)\n", lo_inst->getInstId(), i, j);
                    l_error_count++; 
                } 
                lo_inst_attend.insert(lo_inst);
            }
        } 
    }
    return l_error_count;
}

// 模拟退火算法实现
void Solution::simulatedAnnealing() {
    // 初始化随机数生成器
    std::random_device rd;
    std::mt19937 gen(rd());
    
    // 初始温度、终止温度和冷却率
    double temperature = 10000.0;
    double finalTemperature = 0.1;
    double coolingRate = 0.95;
    
    // 收集未固定的Instance
    std::vector<Instance*> movableInsts;
    for (auto& inst_pair : glb_inst_map) {
        if (!inst_pair.second->isFixed()) {
            movableInsts.push_back(inst_pair.second);
        }
    }
    
    // 如果没有可移动的Instance，直接返回
    if (movableInsts.empty()) {
        std::cout << "No movable instances found." << std::endl;
        return;
    }
    
    // 创建FPGA占用状态矩阵
    std::vector<std::vector<bool>> occupied(
        glb_fpga.getSizeX(), 
        std::vector<bool>(glb_fpga.getSizeY(), false)
    );
    
    // 标记已占用的位置
    for (int i = 0; i < glb_fpga.getSizeX(); i++) {
        for (int j = 0; j < glb_fpga.getSizeY(); j++) {
            Block* block = glb_fpga.getBlock(i, j);
            if (block && block->getInstsCount() > 0) {
                occupied[i][j] = true;
            }
        }
    }
    
    // 初始布局：将可移动的Instance随机放置在空闲位置
    std::vector<std::pair<int, int>> emptyPositions;
    for (int i = 0; i < glb_fpga.getSizeX(); i++) {
        for (int j = 0; j < glb_fpga.getSizeY(); j++) {
            if (!occupied[i][j]) {
                emptyPositions.push_back({i, j});
            }
        }
    }
    
    // 如果空位不足，报错并退出
    if (emptyPositions.size() < movableInsts.size()) {
        std::cout << "Error: Not enough empty positions for all movable instances." << std::endl;
        return;
    }
    
    // 随机打乱空位顺序
    std::shuffle(emptyPositions.begin(), emptyPositions.end(), gen);
    
    // 初始放置
    for (size_t i = 0; i < movableInsts.size(); i++) {
        int x = emptyPositions[i].first;
        int y = emptyPositions[i].second;
        movableInsts[i]->setPosition(x, y);
        glb_fpga.addInst(x, y, movableInsts[i]);
        occupied[x][y] = true;
    }
    
    // 计算初始总线长度
    int currentWireLength = reportWireLength();
    int bestWireLength = currentWireLength;
    
    // 存储最优解
    std::vector<std::pair<Instance*, std::pair<int, int>>> bestSolution;
    for (auto& inst_pair : glb_inst_map) {
        bestSolution.push_back({inst_pair.second, inst_pair.second->getPosition()});
    }
    
    // 记录迭代次数和接受移动的次数
    int iterations = 0;
    int acceptedMoves = 0;
    
    // 在指定温度范围内迭代
    while (temperature > finalTemperature) {
        iterations++;
        
        // 在每个温度下进行多次尝试
        int innerIterations = 100;
        for (int i = 0; i < innerIterations; i++) {
            // 随机选择一个可移动的Instance
            std::uniform_int_distribution<> instDist(0, movableInsts.size() - 1);
            int instIdx = instDist(gen);
            Instance* inst = movableInsts[instIdx];
            
            // 获取当前位置
            std::pair<int, int> currentPos = inst->getPosition();
            
            // 随机选择一个空闲位置
            std::vector<std::pair<int, int>> availablePositions;
            for (int x = 0; x < glb_fpga.getSizeX(); x++) {
                for (int y = 0; y < glb_fpga.getSizeY(); y++) {
                    // 如果位置未被占用或者就是当前Instance的位置
                    if (!occupied[x][y] || (x == currentPos.first && y == currentPos.second)) {
                        availablePositions.push_back({x, y});
                    }
                }
            }
            
            std::uniform_int_distribution<> posDist(0, availablePositions.size() - 1);
            int posIdx = posDist(gen);
            std::pair<int, int> newPos = availablePositions[posIdx];
            
            // 如果新位置与当前位置相同，则跳过
            if (newPos.first == currentPos.first && newPos.second == currentPos.second) {
                continue;
            }
            
            // 保存当前状态
            occupied[currentPos.first][currentPos.second] = false;
            glb_fpga.clearInst(currentPos.first, currentPos.second);
            
            // 移动到新位置
            inst->setPosition(newPos.first, newPos.second);
            occupied[newPos.first][newPos.second] = true;
            glb_fpga.addInst(newPos.first, newPos.second, inst);
            
            // 计算新的总线长度
            int newWireLength = reportWireLength();
            
            // 计算能量变化
            int deltaE = newWireLength - currentWireLength;
            
            // 决定是否接受此移动
            bool accept = false;
            if (deltaE <= 0) {
                // 如果能量减小，总是接受
                accept = true;
            } else {
                // 如果能量增加，以一定概率接受
                std::uniform_real_distribution<> dist(0.0, 1.0);
                double probability = exp(-deltaE / temperature);
                if (dist(gen) < probability) {
                    accept = true;
                }
            }
            
            if (accept) {
                // 接受新解
                currentWireLength = newWireLength;
                acceptedMoves++;
                
                // 更新最优解
                if (newWireLength < bestWireLength) {
                    bestWireLength = newWireLength;
                    bestSolution.clear();
                    for (auto& inst_pair : glb_inst_map) {
                        bestSolution.push_back({inst_pair.second, inst_pair.second->getPosition()});
                    }
                }
            } else {
                // 不接受，回退到之前的状态
                occupied[newPos.first][newPos.second] = false;
                glb_fpga.clearInst(newPos.first, newPos.second);
                
                inst->setPosition(currentPos.first, currentPos.second);
                occupied[currentPos.first][currentPos.second] = true;
                glb_fpga.addInst(currentPos.first, currentPos.second, inst);
            }
        }
        
        // 降低温度
        temperature *= coolingRate;
        
        // 输出当前温度下的进度
        if (iterations % 10 == 0) {
            std::cout << "Temperature: " << temperature << ", Current WL: " << currentWireLength 
                      << ", Best WL: " << bestWireLength << std::endl;
        }
    }
    
    // 恢复最优解
    for (auto& solution : bestSolution) {
        Instance* inst = solution.first;
        std::pair<int, int> pos = solution.second;
        
        // 如果不是当前位置，需要更新
        std::pair<int, int> currentPos = inst->getPosition();
        if (currentPos.first != pos.first || currentPos.second != pos.second) {
            // 清除当前位置
            occupied[currentPos.first][currentPos.second] = false;
            glb_fpga.clearInst(currentPos.first, currentPos.second);
            
            // 设置到最优位置
            inst->setPosition(pos.first, pos.second);
            occupied[pos.first][pos.second] = true;
            glb_fpga.addInst(pos.first, pos.second, inst);
        }
    }
    
    std::cout << "Simulated annealing completed." << std::endl;
    std::cout << "Total iterations: " << iterations << std::endl;
    std::cout << "Accepted moves: " << acceptedMoves << std::endl;
    std::cout << "Final wirelength: " << bestWireLength << std::endl;
}

// 添加一个辅助函数，清理FPGA上的所有非固定Instance
void Solution::clearNonFixedInstances() {
    for (int i = 0; i < glb_fpga.getSizeX(); i++) {
        for (int j = 0; j < glb_fpga.getSizeY(); j++) {
            Block* block = glb_fpga.getBlock(i, j);
            if (block && !block->getInsts().empty()) {
                Instance* inst = block->getInsts()[0];
                if (inst && !inst->isFixed()) {
                    glb_fpga.clearInst(i, j);
                }
            }
        }
    }
}

// 执行布局优化的主函数
void Solution::runPlacement() {
    std::cout << "Starting placement algorithm..." << std::endl;
    
    // 清除所有非固定Instance的布局
    clearNonFixedInstances();
    
    // 执行模拟退火算法
    simulatedAnnealing();
    
    // 验证最终布局是否合法
    int errors = reportValid();
    if (errors > 0) {
        std::cout << "Warning: Found " << errors << " errors in the final placement." << std::endl;
    } else {
        std::cout << "Placement completed successfully with no errors." << std::endl;
    }
    
    // 报告最终线长
    std::cout << "Final ";
    int finalWL = reportWireLength();
    std::cout << "Placement optimization completed." << std::endl;
}

## 实验报告

### 1. 算法逻辑和实现思路

#### 1.1 算法选择：模拟退火算法

本实验中，我采用了**模拟退火算法**来解决FPGA布局问题。模拟退火算法是一种基于随机搜索的优化算法，它模拟了固体退火过程中原子运动趋于最小能量状态的物理现象

选择模拟退火算法的原因如下：
1. 能够跳出局部最优解，有可能找到全局最优解
2. 适合处理大规模组合优化问题，特别是NP难问题
3. 在VLSI布局问题中有良好的应用历史和实践效果
4. 实现相对简单，参数调整灵活

#### 1.2 算法详细设计

模拟退火算法的核心思想是通过接受一定概率的"坏解"来避免陷入局部最优，随着温度的降低，算法逐渐趋向于只接受"好解"，最终稳定在一个较优解上。

算法流程如下：
1. **初始化**：
   - 设置初始温度、终止温度和冷却率
   - 随机生成初始布局，将可移动元件随机放置在FPGA的空闲Block上
   - 计算初始布局的总线长度（HPWL）

2. **迭代优化**：
   - 在每个温度下进行多次内部迭代
   - 每次随机选择一个可移动元件和一个目标位置
   - 尝试将元件移动到新位置，计算新布局的线长变化ΔE
   - 根据Metropolis准则决定是否接受移动：
     - 若ΔE ≤ 0（线长减少或不变），则接受移动
     - 若ΔE > 0（线长增加），以概率p = exp(-ΔE/T)接受移动
   - 记录并更新最佳解
   - 降低温度T = T * 冷却率

3. **完成布局**：
   - 当温度降到终止温度以下时，停止迭代
   - 恢复记录的最佳布局
   - 验证布局合法性并输出结果

#### 1.3 约束处理

在实现过程中，需要处理以下约束：

1. **固定元件约束**：某些元件是固定的，不能移动
   - 解决方法：在算法中检查元件的isFixed()属性，只移动非固定元件

2. **Block容量约束**：每个Block只能放置一个元件（MAX_BLOCK_CAPACITY = 1）
   - 解决方法：使用occupied矩阵记录FPGA上每个位置的占用情况，确保元件只能移动到空闲位置

3. **Instance和Block的关联**：
   - 解决方法：每次移动元件时，同时更新Instance的位置信息和FPGA的Block内容，保证两者一致

#### 1.4 算法创新点

1. **自适应内部迭代次数**：
   - 在每个温度下，内部迭代次数可以根据可移动元件的数量自动调整，提高效率

2. **优化移动策略**：
   - 不是简单的随机移动，而是倾向于移动到与其连接元件更近的位置
   - 可以通过计算元件连接网络的质心，优先考虑向质心方向移动

3. **记忆机制**：
   - 记录最佳解并在算法结束时恢复，避免最终解未必是全局遍历过的最优解的问题

### 3. 实验总结

#### 3.1 方法优点

1. **全局搜索能力**：模拟退火算法具有跳出局部最优的能力，能够在解空间中进行广泛探索
2. **参数调整灵活**：通过调整初始温度、冷却率等参数，可以平衡算法的搜索能力和收敛速度
3. **适应性强**：可以处理不同规模和复杂度的FPGA布局问题
4. **实现简单**：算法思路清晰，易于实现和调试

#### 3.2 方法缺点

1. **计算开销大**：特别是在大规模问题上，模拟退火需要较长的运行时间
2. **参数敏感**：算法性能很大程度上依赖于初始温度、冷却率等参数的选择
3. **随机性**：由于有随机因素，每次运行结果可能存在差异
4. **收敛慢**：在后期温度较低时，改进速度会明显减慢

#### 3.3 改进方向
